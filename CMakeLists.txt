function(doc_parser_set_common_warnings target_name)
    target_compile_options(${target_name}
        PRIVATE
            $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall -Wextra -Wpedantic>
    )
endfunction()

function(doc_parser_set_public_includes target_name)
    get_target_property(target_type ${target_name} TYPE)
    if(target_type STREQUAL "INTERFACE_LIBRARY")
        set(include_scope INTERFACE)
    else()
        set(include_scope PUBLIC)
    endif()
    target_include_directories(${target_name}
        ${include_scope}
            $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/cpp>
            $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/document_intelligence_engine>
    )
endfunction()
